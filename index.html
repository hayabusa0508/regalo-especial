function crearCorazonesFondo() { const container = document.getElementById('hearts-container'); const simbolos = ['❤️', '💖', '💕', '✨']; setInterval(() => { const heart = document.createElement('div'); heart.className = 'heart'; // Elegir un símbolo aleatorio heart.textContent = simbolos[Math.floor(Math.random() * simbolos.length)]; // Posición horizontal aleatoria heart.style.left = Math.random() * 100 + 'vw'; // Duración de la animación aleatoria (entre 4 y 8 segundos) const duracion = Math.random() * 4 + 4; heart.style.animationDuration = duracion + 's'; // Tamaños variados para dar profundidad const tamano = Math.random() * 0.8 + 0.8; heart.style.fontSize = tamano + 'rem'; container.appendChild(heart); // Limpiar el elemento del DOM al terminar la animación setTimeout(() => { heart.remove(); }, duracion * 1000); }, 400); // Crea un corazón nuevo cada 400ms } // Iniciar los corazones al cargar la página window.addEventListener('DOMContentLoaded', crearCorazonesFondo);
